41 static inline bool hasEvenNoOfBits(uint16_t data)
42 {
43 // Fold upper bits into lower bits; after these XORs, LSB holds parity.
44 data ^= data >> 8;
45 data ^= data >> 4;
46 data ^= data >> 2;
47 data ^= data >> 1;
48
49 // If LSB == 0 ⇒ even parity; if LSB == 1 ⇒ odd parity.
50 return ((~data) & 1u) != 0u;
51 }
